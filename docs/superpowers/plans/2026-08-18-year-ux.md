**Goal:** Add a `‹ 2026 › ⚙` year navigation control to the Activities page with a Manage Years panel (CSV export included), a CEA Import modal with conflict preview, Planning Workspace year badge, and clean up all "Planning Year"/"Training Year" label strings throughout the TMS frontend.

**Architecture:** All frontend changes go in `connected-frontend/index.html` (single-file SPA, no build step). Backend adds one new `GET /api/planning/years/{year_id}/export` endpoint and adds an optional `keep_existing` form field to the existing CEA import endpoint. Activities are NOT year-scoped via FK; `CeaActivity` (has `planning_year_id`) drives both the export activities section and the import flow. No schema migration.

**Tech Stack:** FastAPI + SQLAlchemy + Python stdlib `csv`/`io` (backend), vanilla JS + inline CSS (frontend), Playwright (E2E tests, staging).

```python
@router.get("/years/{year_id}/export")
async def export_year_csv(
    year_id: str,
    db: DBSession = Depends(get_db),
    p: Principal = Depends(get_principal),
):
    """Download a planning year's full program as a CSV file."""
    import io
    from datetime import date as _date

    year = db.query(PlanningYear).filter(PlanningYear.id == year_id).first()
    if not year:
        raise HTTPException(404, "Year not found")
    _require_year_access(p, year)

    # Determine scope label for filename and header
    if hasattr(p, "squadron_name") and p.squadron_name:
        scope_label = p.squadron_name.upper()
    elif hasattr(p, "wing_name") and p.wing_name:
        scope_label = p.wing_name.upper()
    else:
        scope_label = "NATIONAL"

    buf = io.StringIO()
    writer = _csv.writer(buf)

    # ── Header ──────────────────────────────────────────────────────
    writer.writerow([f"TRAINING PROGRAM — {scope_label} — {year.year}"])
    writer.writerow([f"Exported: {_date.today().isoformat()}"])
    writer.writerow([])

    # ── CEA Activities ───────────────────────────────────────────────
    # CeaActivity has planning_year_id; Activity (training.py) does not.
    writer.writerow(["ACTIVITIES"])
    writer.writerow(["cea_activity_id", "name", "type", "importance", "owning_level"])
    activities = (
        db.query(CeaActivity)
        .filter(CeaActivity.planning_year_id == year_id, CeaActivity.is_archived.is_(False))
        .order_by(CeaActivity.activity_name)
        .all()
    )
    for act in activities:
        writer.writerow([
            act.cea_activity_id or "",
            act.activity_name or "",
            act.activity_type or "",
            act.importance or "",
            act.parent_unit or act.host_unit or "",
        ])
    writer.writerow([])

    # ── Parade Schedule ──────────────────────────────────────────────
    writer.writerow(["PARADE SCHEDULE"])
    writer.writerow(["date", "type", "term", "week", "notes"])
    parade_dates = (
        db.query(ParadeDate)
        .filter(ParadeDate.planning_year_id == year_id, ParadeDate.is_active.is_(True))
        .order_by(ParadeDate.parade_date)
        .all()
    )
    for pd in parade_dates:
        writer.writerow([
            pd.parade_date or "",
            pd.parade_type or "",
            pd.term or "",
            pd.week_number or "",
            (pd.notes or "").replace("\n", " "),
        ])
    writer.writerow([])

    # ── Sessions ─────────────────────────────────────────────────────
    # Session → parade_night_id → ParadeNight ← ParadeDate.parade_night_id
    writer.writerow(["SESSIONS"])
    writer.writerow(["date", "period", "title", "facilitator", "training_classes"])
    sessions = (
        db.query(TrainingSession)
        .join(ParadeNight, TrainingSession.parade_night_id == ParadeNight.id)
        .join(ParadeDate, ParadeDate.parade_night_id == ParadeNight.id)
        .filter(ParadeDate.planning_year_id == year_id)
        .order_by(ParadeDate.parade_date, TrainingSession.period_number)
        .all()
    )
    for sess in sessions:
        parade_date_str = ""
        if sess.parade_night and sess.parade_night.parade_dates:
            pd0 = sess.parade_night.parade_dates[0]
            parade_date_str = pd0.parade_date or ""
        classes = "; ".join(
            sa.training_class.name
            for sa in (sess.session_audiences or [])
            if sa.training_class
        )
        writer.writerow([
            parade_date_str,
            sess.period_number or "",
            sess.custom_title or sess.curriculum_title_at_time or "",
            sess.facilitator_display_name_at_time or "",
            classes,
        ])

    csv_bytes = buf.getvalue().encode("utf-8-sig")  # BOM for Excel compatibility
    safe_scope = scope_label.replace(" ", "_").replace("/", "-")
    filename = f"AAFC_TMS_{year.year}_{safe_scope}_{_date.today().isoformat()}.csv"

    return StreamingResponse(
        io.BytesIO(csv_bytes),
        media_type="text/csv",
        headers={"Content-Disposition": f'attachment; filename="{filename}"'},
    )
```

```javascript
// ── Year Nav Control ─────────────────────────────────────────────────────────
// P.years        Array<{id,year,active_status}> sorted by year asc
// P.currentYearId  UUID of selected year
// P.currentYearInt number (e.g. 2026)
// P._ynEditing   bool — true while inline input is open

async function ynInit(){
  let years;
  try{ years=await api('/planning/years'); }
  catch(e){ console.warn('ynInit: could not fetch years',e); return; }
  P.years=(years||[]).sort((a,b)=>a.year-b.year);

  let initial=null;
  if(P.currentYearId){
    initial=P.years.find(y=>y.id===P.currentYearId)||null;
  }
  if(!initial){
    const active=P.years.filter(y=>y.active_status);
    initial=active.length?active[active.length-1]:P.years[P.years.length-1]||null;
  }
  if(initial) setCurrentYear(initial);

  // Show gear for admin roles
  const adminRoles=['system_admin','wing_admin','national_admin'];
  const gear=document.getElementById('ynGear');
  if(gear&&adminRoles.includes(S.role)) gear.style.display='';
}

function setCurrentYear(yearObj){
  P.currentYearId=yearObj.id;
  P.currentYearInt=yearObj.year;

  // Update year nav label
  const lbl=document.getElementById('ynLabel');
  if(lbl) lbl.textContent=yearObj.year;

  // Update arrow disabled states
  const idx=P.years.findIndex(y=>y.id===yearObj.id);
  const prev=document.getElementById('ynPrev'), next=document.getElementById('ynNext');
  if(prev) prev.disabled=(idx<=0);
  if(next) next.disabled=(idx>=P.years.length-1);

  // Sync PW badge (Task 5 wires these elements)
  const badge=document.getElementById('navPwYrBadge');
  if(badge){ badge.textContent=yearObj.year; badge.style.display=''; }
  const hint=document.getElementById('navPwYrHint');
  if(hint){ hint.textContent='Opens with '+yearObj.year+' context'; hint.style.display=''; }

  // Reload Activities page data (year used as context for new activities and imports)
  if(typeof _loadActivitiesPage==='function') _loadActivitiesPage();
}

function ynNav(dir){
  if(P._ynEditing) return;
  const idx=P.years.findIndex(y=>y.id===P.currentYearId);
  const n=idx+dir;
  if(n<0||n>=P.years.length) return;
  setCurrentYear(P.years[n]);
}

function ynStartEdit(){
```

```javascript
async function ceaPreview(){

  if(!fi||!fi.files||!fi.files.length) return;
  const errEl=document.getElementById('ceaUploadErr');
  errEl.textContent='';
  const btn=document.getElementById('ceaPreviewBtn');
  btn.disabled=true; btn.textContent='Reading…';

  // Read file as text
  const file=fi.files[0];
  let text;
  try{
    text=await new Promise(function(resolve,reject){
      const r=new FileReader();
      r.onload=function(e){resolve(e.target.result);};
      r.onerror=function(){reject(new Error('Could not read file'));};
      r.readAsText(file,'UTF-8');
    });
  } catch(e){
    errEl.textContent='Could not read file — check it is a valid CSV.';
    btn.disabled=false; btn.textContent='Preview import';
    return;
  }

  // Simple CSV parse (handles quoted fields)
  function parseCSV(str){
    const lines=str.split(/\r?\n/).filter(function(l){return l.trim();});
    if(!lines.length) return [];
    const headers=lines[0].split(',').map(function(h){return h.trim().replace(/^"|"$/g,'');});
    return lines.slice(1).map(function(line){
      const vals=[]; let cur=''; let inQ=false;
      for(let i=0;i<line.length;i++){
        const ch=line[i];
        if(ch==='"'){inQ=!inQ;}
        else if(ch===','&&!inQ){vals.push(cur.trim());cur='';}
        else{cur+=ch;}
      }
      vals.push(cur.trim());
      const obj={};
      headers.forEach(function(h,i){obj[h]=vals[i]||'';});
      return obj;
    });
  }

  try{ _ceaRows=parseCSV(text); }
  catch(e){ errEl.textContent='Parse error: '+e.message; btn.disabled=false; btn.textContent='Preview import'; return; }

  if(!_ceaRows.length){ errEl.textContent='No rows found in file.'; btn.disabled=false; btn.textContent='Preview import'; return; }

    const existing=await api('/planning/years/'+P.currentYearId+'/cea-activities');

    // If endpoint unavailable, assume empty (no conflicts detectable)
    _ceaExisting=[];
  }

  // Categorise rows
  // CEA file columns: try common aliases for the ID field
  const ID_ALIASES=['ActivityID','cea_activity_id','ActivityId','SeqNr','CEA_ID'];
  const NAME_ALIASES=['ActivityName','activity_name','Name'];
  function getField(row,aliases){
    for(const k of aliases){ if(row[k]!==undefined) return row[k]||''; }
    return Object.values(row)[0]||'';
  }

  const existingById={};

  if(counts.new) html+='<p style="font-size:12px;color:var(--ok,#1a7f4b);font-weight:700;margin-bottom:6px">'+counts.new+' new activities will be added</p>';
  if(counts.update) html+='<p style="font-size:12px;color:var(--royal);font-weight:700;margin-bottom:6px">'+counts.update+' existing activities will be updated</p>';
  if(counts.conflict) html+='<p style="font-size:12px;color:#c97a00;font-weight:700;margin-bottom:6px">'+counts.conflict+' conflicts with squadron-imported activities — review below</p>';

  html+='<div style="overflow-x:auto;max-height:300px;overflow-y:auto;border:1px solid var(--border);border-radius:6px">'
    +'<table class="yn-mgmt-table"><thead><tr><th>CEA ID</th><th>Name</th><th>Status</th><th>Decision</th></tr></thead><tbody>';

  categorised.forEach(function(c,i){
    const rowCls=c.status==='new'?'cea-row-new':c.status==='update'?'cea-row-update':'cea-row-conflict';
    const statusLbl=c.status==='new'?'<span style="color:var(--ok,#1a7f4b);font-weight:700">New</span>'
      :c.status==='update'?'<span style="color:var(--royal);font-weight:700">Update</span>'
      :'<span style="color:#c97a00;font-weight:700">Conflict</span>';
    const decision=c.status==='conflict'
      ?'<button class="cea-conflict-toggle replace" id="ceatog-'+i+'" onclick="ceaToggleKeep(this,\''+c.ceaId+'\')">Replace</button>'
      :'—';
    html+='<tr class="'+rowCls+'"><td style="font-size:11px;color:var(--muted)">'+esc(c.ceaId)+'</td><td style="font-weight:600">'+esc(c.name)+'</td><td>'+statusLbl+'</td><td>'+decision+'</td></tr>';
  });

  html+='</tbody></table></div>';

  document.getElementById('ceaPreviewBody').innerHTML=html;
  document.getElementById('ceaImportCount').textContent=_ceaRows.length;
  document.getElementById('ceaStep1').style.display='none';
  document.getElementById('ceaStep2').style.display='';
  btn.disabled=false; btn.textContent='Preview import';
}

function ceaToggleKeep(btn,ceaId){
  if(_ceaKeepIds.has(ceaId)){
    _ceaKeepIds.delete(ceaId);
    btn.textContent='Replace';
    btn.className='cea-conflict-toggle replace';
  } else {
    _ceaKeepIds.add(ceaId);
    btn.textContent='Keep existing';
    btn.className='cea-conflict-toggle keep';
  }
}

  document.getElementById('ceaStep1').style.display='';
}

async function ceaConfirm(){
  const fi=document.getElementById('ceaFileInput');
  if(!fi||!fi.files||!fi.files.length){ alert('No file selected.'); return; }
  const btn=document.querySelector('#ceaStep2 .btn-dk');
  btn.disabled=true; btn.textContent='Importing…';

  const formData=new FormData();
  formData.append('file',fi.files[0]);
  if(_ceaKeepIds.size) formData.append('keep_existing',Array.from(_ceaKeepIds).join(','));

  try{
    const tok=sessionStorage.getItem('aafc_token')||'';
    const resp=await fetch(API_BASE+'/planning/years/'+P.currentYearId+'/cea/import',{
      method:'POST',
      headers:tok?{Authorization:'Bearer '+tok}:{},
      credentials:'include',
      body:formData,
    });
    if(!resp.ok){
      const err=await resp.json().catch(function(){return {};});
      throw new Error(err.detail||'Import failed ('+resp.status+')');
    }
    const result=await resp.json();
    const created=result.created||0, updated=result.updated||0, skipped=result.skipped||0;
    document.getElementById('ceaResultBody').innerHTML=
      '<div style="font-size:14px;font-weight:700;color:var(--ok,#1a7f4b);margin-bottom:8px">Import complete</div>'
      +'<p style="font-size:13px">'+created+' added &nbsp;·&nbsp; '+updated+' updated &nbsp;·&nbsp; '+skipped+' kept existing</p>';
    document.getElementById('ceaStep2').style.display='none';
    document.getElementById('ceaStep3').style.display='';
  } catch(e){
    btn.disabled=false; btn.textContent='Import all ('+_ceaRows.length+')';
    document.getElementById('ceaUploadErr').textContent='';
    const errEl=document.getElementById('ceaPreviewBody');
    errEl.insertAdjacentHTML('afterbegin','<div style="color:#c01530;font-size:12px;font-weight:700;margin-bottom:8px">Import failed: '+esc(e.message)+'</div>');
  }
}
```
